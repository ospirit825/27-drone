cmake_minimum_required(VERSION 3.16)
project(armor_detector LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
find_package(OpenCV 4.5.2 REQUIRED COMPONENTS core imgproc imgcodecs videoio)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
find_package(nlohmann_json 3.9 REQUIRED)
set(CVI_SDK_ROOT "" CACHE PATH "SDK root containing include/cviruntime.h and lib/libcviruntime.so")
find_path(CVI_INCLUDE_DIR cviruntime.h HINTS "${CVI_SDK_ROOT}/include" REQUIRED)
find_library(CVI_LIBRARY cviruntime HINTS "${CVI_SDK_ROOT}/lib" /usr/bin/lib /mnt/system/usr/lib REQUIRED)
add_executable(armor_detector armor_detector.cpp)
target_include_directories(armor_detector PRIVATE ${CVI_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(armor_detector PRIVATE ${CVI_LIBRARY} ${OpenCV_LIBS} Threads::Threads ZLIB::ZLIB nlohmann_json::nlohmann_json)
target_compile_options(armor_detector PRIVATE -O3 -Wall -Wextra)
option(ARMOR_BUILD_CHECK "Build preprocessing/postprocessing regression check" OFF)
if(ARMOR_BUILD_CHECK)
  add_executable(armor_check check.cpp)
  target_include_directories(armor_check PRIVATE ${CVI_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(armor_check PRIVATE ${CVI_LIBRARY} ${OpenCV_LIBS} Threads::Threads ZLIB::ZLIB nlohmann_json::nlohmann_json)
  enable_testing()
  add_test(NAME armor_check COMMAND armor_check)
endif()
